-- ===================== SAFE GUI PARENT (Delta/executor hardening) =====================
local function getSafeGuiParent()
    local playerGui = nil
    pcall(function()
        playerGui = LocalPlayer:WaitForChild("PlayerGui", 5)
    end)
    if playerGui then
        return playerGui
    end

    local success, parent = pcall(function()
        if type(gethui) == "function" then
            return gethui()
        end
        return game:GetService("CoreGui")
    end)
    if success and parent then
        return parent
    end

    return nil
end

local targetParent = getSafeGuiParent()
if not targetParent then
    warn("[Rex] Could not find a safe GUI parent — aborting.")
    return
end

for _, container in ipairs({LocalPlayer:FindFirstChildOfClass("PlayerGui"), targetParent}) do
    if container then
        pcall(function()
            for _, child in ipairs(container:GetChildren()) do
                if child.Name == "RexScript" or child.Name == "AXN" then child:Destroy() end
            end
        end)
    end
end

-- ===================== ENFORCEMENT LOOPS =====================
RunService:BindToRenderStep("REX_SpeedJumpEnforce", Enum.RenderPriority.Last.Value + 1, function()
    if not (Humanoid and Humanoid.Parent) then return end

    -- ---- Speed ----
    local targetSpeed = speedEnabled and currentSpeed or 16
    if Humanoid.WalkSpeed ~= targetSpeed then
        Humanoid.WalkSpeed = targetSpeed
    end
    if speedEnabled then
        local rootPart = Character and Character:FindFirstChild("HumanoidRootPart")
        if rootPart and Humanoid.MoveDirection.Magnitude > 0 then
            local horizontal = Humanoid.MoveDirection * currentSpeed
            local vel = rootPart.AssemblyLinearVelocity
            rootPart.AssemblyLinearVelocity = Vector3.new(horizontal.X, vel.Y, horizontal.Z)
        end
    end

    -- ---- Jump (property layer; JumpRequest hook below is the main layer) ----
    local targetJump = jumpEnabled and currentJump or DEFAULT_JUMP_POWER
    local targetHeight = (targetJump / DEFAULT_JUMP_POWER) * 7.2
    if Humanoid.JumpPower ~= targetJump then
        Humanoid.JumpPower = targetJump
    end
    if math.abs(Humanoid.JumpHeight - targetHeight) > 0.01 then
        Humanoid.JumpHeight = targetHeight
    end
end)

-- Direct-velocity jump boost on the exact moment of takeoff.
local jumpRequestConn = UserInputService.JumpRequest:Connect(function()
    if not jumpEnabled then return end
    local rootPart = Character and Character:FindFirstChild("HumanoidRootPart")
    if rootPart then
        local vel = rootPart.AssemblyLinearVelocity
        rootPart.AssemblyLinearVelocity = Vector3.new(vel.X, currentJump, vel.Z)
    end
end)
table.insert(activeConnections, jumpRequestConn)

-- Helper: full-width drag slider, parented into SlidersContainer
local function createSlider(labelPrefix, order, minVal, maxVal, startVal, onChange)
    local label = Instance.new("TextLabel")
    label.Name = labelPrefix .. "Label"
    label.LayoutOrder = order
    label.Size = UDim2.new(1, 0, 0, 18)
    label.BackgroundTransparency = 1
    label.Text = labelPrefix .. ": " .. startVal
    label.TextColor3 = Color3.fromRGB(200, 225, 255)
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Visible = false
    label.Parent = SlidersContainer

    local track = Instance.new("Frame")
    track.Name = labelPrefix .. "Track"
    track.LayoutOrder = order + 1
    track.Size = UDim2.new(1, 0, 0, 10)
    track.BackgroundColor3 = Color3.fromRGB(20, 23, 32)
    track.BorderSizePixel = 0
    track.Active = true
    track.Visible = false
    track.Parent = SlidersContainer

    local trackCorner = Instance.new("UICorner")
    trackCorner.CornerRadius = UDim.new(1, 0)
    trackCorner.Parent = track

    local fill = Instance.new("Frame")
    fill.Name = "Fill"
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = NEON_BLUE
    fill.BorderSizePixel = 0
    fill.Parent = track

    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill

    local knob = Instance.new("Frame")
    knob.Name = "Knob"
    knob.Size = UDim2.new(0, 16, 0, 16)
    knob.AnchorPoint = Vector2.new(0.5, 0.5)
    knob.Position = UDim2.new(0, 0, 0.5, 0)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel = 0
    knob.ZIndex = 2
    knob.Active = true
    knob.Parent = track

    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = knob

    local dragging = false

    local function setFromX(xPos)
        local trackAbsPos = track.AbsolutePosition.X
        local trackAbsSize = track.AbsoluteSize.X
        local rawAlpha = (xPos - trackAbsPos) / trackAbsSize
        local alpha = math.clamp(rawAlpha, 0, 1)
        local value = math.floor(minVal + (maxVal - minVal) * alpha)

        fill.Size = UDim2.new(alpha, 0, 1, 0)
        knob.Position = UDim2.new(alpha, 0, 0.5, 0)
        label.Text = labelPrefix .. ": " .. value

        onChange(value)
    end

    local function beginDrag(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end

    track.InputBegan:Connect(beginDrag)
    knob.InputBegan:Connect(beginDrag)

    local changedConn = UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    table.insert(activeConnections, changedConn)

    local endedConn = UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    table.insert(activeConnections, endedConn)

    task.defer(function()
        RunService.Heartbeat:Wait()
        setFromX(track.AbsolutePosition.X + (track.AbsoluteSize.X * ((startVal - minVal) / (maxVal - minVal))))
    end)

    local api = {}
    function api.SetVisible(visible)
        label.Visible = visible
        track.Visible = visible
    end

    return api
end

local function setSoundMuted(sound, muted)
    if muted then
        if mutedSounds[sound] == nil then mutedSounds[sound] = sound.Volume end
        sound.Volume = 0
    elseif mutedSounds[sound] ~= nil then
        pcall(function() sound.Volume = mutedSounds[sound] end)
        mutedSounds[sound] = nil
    end
end

MuteBtn.MouseButton1Click:Connect(function()
    muteEnabled = not muteEnabled
    if muteEnabled then
        for _, obj in ipairs(game:GetDescendants()) do
            if obj:IsA("Sound") then setSoundMuted(obj, true) end
        end
        muteWatchConn = game.DescendantAdded:Connect(function(obj)
            if muteEnabled and obj:IsA("Sound") then setSoundMuted(obj, true) end
        end)
        table.insert(activeConnections, muteWatchConn)
    else
        for sound in pairs(mutedSounds) do setSoundMuted(sound, false) end
        mutedSounds = {}
        if muteWatchConn then muteWatchConn:Disconnect(); muteWatchConn = nil end
    end
    MuteBtn.Text = muteEnabled and "Mute: ON" or "Mute: OFF"
    setToggleVisual(MuteBtn, muteEnabled)
end)

-- ===================== FIX LAG (ultra, one-shot) =====================
local function stripEffectsFrom(root)
    local removed = 0
    for _, obj in ipairs(root:GetDescendants()) do
        if EFFECT_CLASSES[obj.ClassName] then
            obj:Destroy()
            removed += 1
        elseif obj:IsA("BasePart") then
            obj.CastShadow = false
        elseif obj:IsA("PointLight") or obj:IsA("SpotLight") or obj:IsA("SurfaceLight") then
            obj.Shadows = false
            obj.Range = 0
        end
    end
    return removed
end

local function applyLightingWipe()
    for _, obj in ipairs(Lighting:GetChildren()) do
        if obj:IsA("BlurEffect") or obj:IsA("ColorCorrectionEffect")
            or obj:IsA("SunRaysEffect") or obj:IsA("DepthOfFieldEffect")
            or obj:IsA("BloomEffect") then
            obj:Destroy()
        elseif obj:IsA("Atmosphere") or obj:IsA("Clouds") then
            obj:Destroy()
        end
    end

    Lighting.GlobalShadows = false
    Lighting.FogEnd = 100000
    pcall(function() Lighting.Technology = Enum.Technology.Compatibility end)

    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        settings().Rendering.EnableFRM = false
    end)

    local Terrain = workspace:FindFirstChildOfClass("Terrain")
    if Terrain then
        Terrain.WaterWaveSize = 0
        Terrain.WaterWaveSpeed = 0
        Terrain.WaterReflectance = 0
        Terrain.WaterTransparency = 0.5
        Terrain.Decoration = false
    end
end

local function fixLag()
    local removed = stripEffectsFrom(workspace)
    applyLightingWipe()

    if not fixLagOn then
        fixLagOn = true
        autoCleanConn = workspace.DescendantAdded:Connect(function(obj)
            if EFFECT_CLASSES[obj.ClassName] then
                obj:Destroy()
            elseif obj:IsA("BasePart") then
                obj.CastShadow = false
            elseif obj:IsA("PointLight") or obj:IsA("SpotLight") or obj:IsA("SurfaceLight") then
                obj.Shadows = false
                obj.Range = 0
            end
        end)
        table.insert(activeConnections, autoCleanConn)
    end

    FixLagBtn.Text = "Đã dọn!"
    task.delay(1, function()
        if FixLagBtn and FixLagBtn.Parent then
            FixLagBtn.Text = "Fix Lag"
        end
    end)
    print("[Rex] Fix Lag applied. Removed " .. removed .. " effect instance(s); auto-clean active.")
end

FixLagBtn.MouseButton1Click:Connect(fixLag)

-- ===================== CUSTOM DRAG (UIScale-aware) =====================
local dragging = false
local dragStartInputPos = nil
local frameStartPos = nil
local totalMoved = 0
local DRAG_THRESHOLD = 6

local function updateDragPosition(inputPos)
    local delta = inputPos - dragStartInputPos
    local scale = MenuScale.Scale
    if scale <= 0 then scale = 1 end
    MainFrame.Position = UDim2.new(
        frameStartPos.X.Scale, frameStartPos.X.Offset + delta.X / scale,
        frameStartPos.Y.Scale, frameStartPos.Y.Offset + delta.Y / scale
    )
    totalMoved = delta.Magnitude
end

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStartInputPos = Vector2.new(input.Position.X, input.Position.Y)
        frameStartPos = MainFrame.Position
        totalMoved = 0
    end
end)

local dragChangedConn = UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
        updateDragPosition(Vector2.new(input.Position.X, input.Position.Y))
    end
end)
table.insert(activeConnections, dragChangedConn)

local dragEndedConn = UserInputService.InputEnded:Connect(function(input)
    if not dragging then return end
    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    dragging = false
    if minimized and totalMoved < DRAG_THRESHOLD then
        toggleMinimize()
    end
end)
table.insert(activeConnections, dragEndedConn)

    end
